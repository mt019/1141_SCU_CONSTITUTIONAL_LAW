\newcommand{\papertexture}{%
  \foreach \i in {1,...,86} {
    \pgfmathsetmacro{\x}{2+104*rnd}
    \pgfmathsetmacro{\y}{4+127*rnd}
    \pgfmathsetmacro{\len}{7+30*rnd}
    \pgfmathsetmacro{\ang}{-12+24*rnd}
    \draw[paperdeep!58!paper,line width=0.13pt]
      (\x mm,\y mm) -- ++(\ang:\len mm);
  }
  \foreach \i in {1,...,24} {
    \pgfmathsetmacro{\x}{4+100*rnd}
    \pgfmathsetmacro{\y}{5+125*rnd}
    \pgfmathsetmacro{\len}{32+55*rnd}
    \pgfmathsetmacro{\ang}{-8+16*rnd}
    \draw[warmline!55!paper,line width=0.25pt]
      (\x mm,\y mm) .. controls ++(\ang+10:13mm) and ++(\ang-15:-16mm) .. ++(\ang:\len mm);
  }
  \foreach \i in {1,...,150} {
    \pgfmathsetmacro{\x}{3+102*rnd}
    \pgfmathsetmacro{\y}{3+129*rnd}
    \fill[dot!58!paper] (\x mm,\y mm) circle[radius=0.055mm];
  }
}

\newcommand{\cardframe}[2]{%
\begin{tikzpicture}[remember picture,overlay]
  \fill[#1] (current page.south west) rectangle (current page.north east);
  \begin{scope}[shift={(current page.south west)}]
    \papertexture
  \end{scope}
  \draw[#2,line width=0.55pt]
    ([xshift=6mm,yshift=6mm]current page.south west)
    rectangle
    ([xshift=-6mm,yshift=-6mm]current page.north east);
  \draw[#2!68!#1,line width=0.25pt]
    ([xshift=8.5mm,yshift=8.5mm]current page.south west)
    rectangle
    ([xshift=-8.5mm,yshift=-8.5mm]current page.north east);
\end{tikzpicture}
}
