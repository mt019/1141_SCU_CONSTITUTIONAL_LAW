\newcommand{\papertexture}{%
  \foreach \i in {1,...,70} {
    \pgfmathsetmacro{\x}{2+104*rnd}
    \pgfmathsetmacro{\y}{4+127*rnd}
    \pgfmathsetmacro{\len}{8+22*rnd}
    \pgfmathsetmacro{\ang}{-18+36*rnd}
    \draw[fiber, opacity=0.16, line width=0.12pt]
      (\x mm,\y mm) -- ++(\ang:\len mm);
  }
  \foreach \i in {1,...,18} {
    \pgfmathsetmacro{\x}{4+100*rnd}
    \pgfmathsetmacro{\y}{5+125*rnd}
    \pgfmathsetmacro{\len}{34+58*rnd}
    \pgfmathsetmacro{\ang}{-8+16*rnd}
    \draw[crease, opacity=0.08, line width=0.35pt]
      (\x mm,\y mm) .. controls ++(\ang+12:12mm) and ++(\ang-18:-18mm) .. ++(\ang:\len mm);
  }
  \foreach \i in {1,...,95} {
    \pgfmathsetmacro{\x}{3+102*rnd}
    \pgfmathsetmacro{\y}{3+129*rnd}
    \fill[ink, opacity=0.035] (\x mm,\y mm) circle[radius=0.08mm];
  }
}
